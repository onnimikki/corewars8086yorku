    + nextOpcode()
}

    - m_ax : short
    - m_bx : short
    - m_cx : short
    - m_dx : short

    - m_ds : short
    - m_es : short
    - m_si : short
    - m_di : short

    - m_ss : short
    - m_bp : short
    - m_sp : short

    - m_cs : short
    - m_ip : short
    - m_flags : short

    - m_energy : short
    - m_bomb1count : byte
    - m_bomb2count : byte

    + getAX() : short
    + getBX() : short
    + getCX() : short
    + getDX() : short

    + setAX(short)
    + setBX(short)
    + setCX(short)
    + setDX(short)

    + getAL() : byte
    + getAH() : byte
    + getBL() : byte
    + getBH() : byte
    + getCL() : byte
    + getCH() : byte
    + getDL() : byte
    + getDH() : byte

    + setAL(byte)
    + setAH(byte)

    + getFlags() : short
    + setFlags(short)
}

class OpcodeFetcher {
    - m_state : CpuState
    - m_memory : RealModeMemory

    + OpcodeFetcher(state : CpuState, memory : RealModeMemory)
    + nextByte() : byte
    + nextWord() : short
}

class RegisterIndexingDecoder {
    - m_state : CpuState

    + getReg8(index : byte) : byte
    + setReg8(index : byte, value : byte)

    + getReg16(index : byte) : short
    + setReg16(index : byte, value : short)

    + getSeg(index : byte) : short
    + setSeg(index : byte, value : short)
}

class IndirectAddressingDecoder {
    - m_state : CpuState
    - m_memory : RealModeMemory
    - m_fetcher : OpcodeFetcher
    - m_regs : RegisterIndexingDecoder

    - m_regIndex : byte
    - m_memIndex : byte
    - m_memAddress : RealModeAddress

    + reset()

    + getRegIndex() : byte
    + getMemAddress() : RealModeAddress

    + getReg8() : byte
    + getMem8() : byte

    + getReg16() : short
    + getMem16() : short

    + getSeg() : short

    + setReg8(byte)
    + setMem8(byte)

    + setReg16(short)
    + setMem16(short)

    + setSeg(short)
}
